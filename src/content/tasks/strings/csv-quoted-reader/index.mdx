Выгрузку из Excel или 1С нужно разобрать, а `strings.Split(line, ",")`
ломается на первом же «Иванов, Иван». Напишите `ReadCSV` сами — без
`encoding/csv`: это частая live-coding задача на конечный автомат. Правила
— в комментарии к функции.

Что проверяют скрытые тесты:

- `"\r\n"` и `"\n"`, пустые строки между записями, последняя запись без
  перевода строки, пустые поля и запятая в конце;
- поле в кавычках с запятой, с `""` внутри и с переводом строки внутри
  (`"\r\n"` в кавычках становится `"\n"`);
- кавычка посреди обычного поля, мусор после закрывающей кавычки,
  незакрытая кавычка — `ErrQuote`;
- разное число полей — `ErrFieldCount`;
- номер строки в ошибке — строка начала записи, с учётом многострочных
  полей выше;
- reader, который отдаёт по одному байту (руна и `""` разрезаны между
  вызовами `Read`), и reader, который ломается посреди данных.

```go
var (
	// ErrQuote — кавычка не на своём месте или не закрыта.
	ErrQuote = errors.New("ошибка кавычек")
	// ErrFieldCount — в записи не столько полей, сколько в первой.
	ErrFieldCount = errors.New("неверное число полей")
)

type CSVError struct {
	Line int
	Err  error
}

func (e *CSVError) Error() string { return fmt.Sprintf("csv, строка %d: %v", e.Line, e.Err) }
func (e *CSVError) Unwrap() error { return e.Err }

func ReadCSV(r io.Reader) ([][]string, error) {
	br := bufio.NewReader(r)
	var records [][]string
	line := 1
	for {
		// Пропускаем пустые строки между записями.
		c, _, err := br.ReadRune()
		for err == nil && (c == '\n' || c == '\r') {
			if c == '\n' {
				line++
			}
			c, _, err = br.ReadRune()
		}
		if err == io.EOF {
			return records, nil
		}
		if err != nil {
			return nil, fmt.Errorf("чтение csv: %w", err)
		}
		if uerr := br.UnreadRune(); uerr != nil {
			return nil, uerr
		}
		start := line
		rec, err := readRecord(br, &line)
		if err != nil {
			var ce *CSVError
			if errors.As(err, &ce) {
				ce.Line = start
			}
			return nil, err
		}
		if len(records) > 0 && len(rec) != len(records[0]) {
			return nil, &CSVError{Line: start, Err: ErrFieldCount}
		}
		records = append(records, rec)
	}
}

func readRecord(br *bufio.Reader, line *int) ([]string, error) {
	var rec []string
	var f strings.Builder
	for {
		quoted := false
		c, _, err := br.ReadRune()
		if err != nil && err != io.EOF {
			return nil, fmt.Errorf("чтение csv: %w", err)
		}
		// Поле в кавычках.
		if err == nil && c == '"' {
			quoted = true
			for {
				c, _, err = br.ReadRune()
				if err == io.EOF {
					return nil, &CSVError{Err: fmt.Errorf("%w: незакрытая кавычка", ErrQuote)}
				}
				if err != nil {
					return nil, fmt.Errorf("чтение csv: %w", err)
				}
				if c == '"' {
					next, _, err2 := br.ReadRune()
					if err2 == nil && next == '"' {
						f.WriteRune('"') // "" внутри кавычек — одна кавычка
						continue
					}
					if err2 == nil {
						_ = br.UnreadRune()
					} else if err2 != io.EOF {
						return nil, fmt.Errorf("чтение csv: %w", err2)
					}
					break
				}
				if c == '\r' {
					// "\r\n" внутри кавычек сохраняем как "\n".
					if next, _, err2 := br.ReadRune(); err2 == nil && next == '\n' {
						c = '\n'
					} else if err2 == nil {
						_ = br.UnreadRune()
					}
				}
				if c == '\n' {
					*line++
				}
				f.WriteRune(c)
			}
			c, _, err = br.ReadRune()
			if err != nil && err != io.EOF {
				return nil, fmt.Errorf("чтение csv: %w", err)
			}
			if err == nil && c == '\r' {
				c, _, err = br.ReadRune()
				if err == nil && c != '\n' {
					return nil, &CSVError{Err: fmt.Errorf("%w: мусор после кавычки", ErrQuote)}
				}
			}
			if err == nil && c != ',' && c != '\n' {
				return nil, &CSVError{Err: fmt.Errorf("%w: мусор после кавычки", ErrQuote)}
			}
		} else {
			// Поле без кавычек — до запятой или конца строки.
			for err == nil && c != ',' && c != '\n' {
				if c == '"' {
					return nil, &CSVError{Err: fmt.Errorf("%w: кавычка внутри поля", ErrQuote)}
				}
				f.WriteRune(c)
				c, _, err = br.ReadRune()
			}
			if err != nil && err != io.EOF {
				return nil, fmt.Errorf("чтение csv: %w", err)
			}
		}
		val := f.String()
		f.Reset()
		rec = append(rec, val)
		if err == io.EOF || c == '\n' {
			if c == '\n' {
				*line++
			}
			if !quoted {
				// "\r\n": '\r' попал в конец поля без кавычек — отрезаем.
				rec[len(rec)-1] = strings.TrimSuffix(val, "\r")
			}
			return rec, nil
		}
		// c == ',' — следующее поле.
	}
}
```

**Запись ≠ строка.** Главная ловушка CSV: перевод строки внутри кавычек —
часть поля. Поэтому читать построчно и резать каждую строку нельзя: нужен
автомат, который знает, внутри он кавычек или нет. Отсюда же номер
строки — считаем каждый `\n`, в том числе внутри полей, а запоминаем номер
в начале записи.

**`bufio.Reader.ReadRune` снимает вопрос о границах `Read`.** Reader
может отдавать данные любыми кусками — хоть по байту. Если разбирать
«что пришло в этом `Read`», двухбайтовая «ё» или пара `""` окажутся
разрезаны. `ReadRune` сам склеивает байты из буфера, а `UnreadRune`
даёт заглянуть на символ вперёд — это нужно для `""` и `"\r\n"`.

**После закрывающей кавычки — только запятая или конец записи.**
`"a"b` — не «ab», а ошибка: иначе испорченный файл тихо превратится в
неправильные данные. Так же строго с кавычкой посреди поля без кавычек.
`encoding/csv` делает то же самое, пока не включён `LazyQuotes`.

**`\r` отрезаем только у полей без кавычек.** В кавычках пользователь мог
сохранить что угодно, а вот `\r` в конце незакавыченного поля — артефакт
Windows. Число полей сверяем с первой записью: колонка, «уехавшая» из-за
лишней запятой, — самый дорогой вид порчи данных, потому что её не видно.
