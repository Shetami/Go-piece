CLI раскрашивает статусы и выводит ссылки, а потом таблица разъезжается:
`len("\x1b[31mошибка\x1b[0m")` — 21 байт, хотя на экране шесть букв.
Напишите три функции: `VisibleWidth`, `PadRight` и `TruncateVisible`.
Какие последовательности бывают — в комментарии в начале файла.

Что проверяют скрытые тесты:

- цвета и стили (CSI с `m`), очистка строки и перемещение курсора (CSI с
  другими финальными байтами), кириллица;
- OSC-гиперссылка с окончанием `ESC \` и заголовок окна с окончанием BEL
  — буквы внутри URL не видимы;
- двухсимвольные последовательности `ESC 7` и `ESC 8`, оборванные CSI и OSC в
  конце строки;
- выравнивание пробелами по видимой ширине;
- обрезка сохраняет стили до места обрезки, отбрасывает стили после
  следующего видимого символа и добавляет сброс, если строка обрезана и
  стили в ней были; без обрезки строка не меняется.

```go
const reset = "\x1b[0m"

func seqEnd(s string, i int) int {
	if i+1 >= len(s) {
		return len(s)
	}
	switch s[i+1] {
	case '[':
		j := i + 2
		for j < len(s) && s[j] >= 0x20 && s[j] <= 0x3F { // параметры и промежуточные
			j++
		}
		if j < len(s) && s[j] >= 0x40 && s[j] <= 0x7E {
			return j + 1
		}
		return len(s) // оборвана или испорчена — до конца
	case ']':
		for j := i + 2; j < len(s); j++ {
			if s[j] == 0x07 {
				return j + 1
			}
			if s[j] == 0x1b && j+1 < len(s) && s[j+1] == '\\' {
				return j + 2
			}
		}
		return len(s)
	default:
		_, size := utf8.DecodeRuneInString(s[i+1:])
		return i + 1 + size
	}
}

func VisibleWidth(s string) int {
	n := 0
	for i := 0; i < len(s); {
		if s[i] == 0x1b {
			i = seqEnd(s, i)
			continue
		}
		_, size := utf8.DecodeRuneInString(s[i:])
		i += size
		n++
	}
	return n
}

func PadRight(s string, width int) string {
	if w := VisibleWidth(s); w < width {
		return s + strings.Repeat(" ", width-w)
	}
	return s
}

func TruncateVisible(s string, width int) string {
	if VisibleWidth(s) <= width {
		return s
	}
	var b strings.Builder
	n, hasSeq := 0, false
	for i := 0; i < len(s); {
		if s[i] == 0x1b {
			j := seqEnd(s, i)
			b.WriteString(s[i:j])
			hasSeq = true
			i = j
			continue
		}
		if n == width {
			break // лимит набран: дальше только последовательности до следующего символа
		}
		_, size := utf8.DecodeRuneInString(s[i:])
		b.WriteString(s[i : i+size])
		i += size
		n++
	}
	if hasSeq {
		b.WriteString(reset)
	}
	return b.String()
}
```

**Одна функция разбора — три потребителя.** `seqEnd` знает только одно:
где кончается последовательность, начатая в этом `ESC`. Ширина, выравнивание
и обрезка строятся поверх неё. Регулярка вида `\x1b\[[0-9;]*m`
ловит только цвета: очистка строки `\x1b[2K`, курсор, OSC-ссылки
останутся «видимыми» и съедят ширину.

**Байты для последовательностей, руны для текста.** Все управляющие
байты — ASCII, поэтому последовательность разбираем по байтам, а видимый
текст — по рунам через `utf8.DecodeRuneInString`. Байт `0x1b` не может
встретиться внутри многобайтовой руны UTF-8 — поэтому смешивать так
безопасно.

**Обрезка не должна ронять стиль.** Если отрезать `\x1b[31mошибка\x1b[0m`
до трёх букв простым срезом по видимым символам, закрывающий `\x1b[0m`
потеряется, и красным станет весь остальной вывод терминала. Поэтому все
последовательности до следующего видимого символа сохраняем, а в конце
добавляем сброс. Заодно `n == width` проверяется только перед видимым
символом — `TruncateVisible(s, 0)` по-прежнему сохраняет открывающий
цвет и корректно его закрывает.

**Ширина 1 — упрощение.** Иероглифы и большинство эмодзи занимают две
клетки, комбинирующие знаки — ноль. В реальных CLI для этого берут
таблицу East Asian Width (`go-runewidth`); на собеседовании достаточно
знать, что «руна = клетка» — не всегда правда.
