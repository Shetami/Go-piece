В колоночном хранилище (так лежат данные в аналитических движках и
DataFrame) строка — это элементы с одним индексом в нескольких слайсах.
`slices.SortFunc` здесь не поможет: он переставляет элементы одного слайса.
А `sort.Interface` — поможет: сортировке нужны только `Len`, `Less` и `Swap`,
а что они делают с данными, её не касается.

Напишите `SortBy` по контракту из заготовки.

Что проверяют скрытые тесты:

- после сортировки по имени возраст и балл остались при своих людях;
- `"-age", "score"`: сначала старшие, внутри возраста — по возрастанию балла;
- `"-score"`: строки с равным баллом **сохраняют исходный порядок** и при
  убывании;
- NaN — в конце и при возрастании, и при убывании, несколько NaN — в исходном
  порядке;
- неизвестный ключ, пустой список ключей, колонки разной длины — ошибка, а
  таблица не тронута.

```go
type Table struct {
	Names  []string
	Ages   []int
	Scores []float64
}

var ErrRagged = errors.New("columns have different lengths")

type sortKey struct {
	col  string
	desc bool
}

type byKeys struct {
	t    *Table
	keys []sortKey
}

func (b byKeys) Len() int { return len(b.t.Names) }

func (b byKeys) Swap(i, j int) {
	t := b.t
	t.Names[i], t.Names[j] = t.Names[j], t.Names[i]
	t.Ages[i], t.Ages[j] = t.Ages[j], t.Ages[i]
	t.Scores[i], t.Scores[j] = t.Scores[j], t.Scores[i]
}

func (b byKeys) Less(i, j int) bool {
	for _, k := range b.keys {
		var c int
		switch k.col {
		case "name":
			c = strings.Compare(b.t.Names[i], b.t.Names[j])
		case "age":
			c = cmp.Compare(b.t.Ages[i], b.t.Ages[j])
		case "score":
			x, y := b.t.Scores[i], b.t.Scores[j]
			xn, yn := math.IsNaN(x), math.IsNaN(y)
			if xn || yn {
				if xn && yn {
					continue
				}
				return yn
			}
			c = cmp.Compare(x, y)
		}
		if k.desc {
			c = -c
		}
		if c != 0 {
			return c < 0
		}
	}
	return false
}

func SortBy(t *Table, keys ...string) error {
	if len(t.Ages) != len(t.Names) || len(t.Scores) != len(t.Names) {
		return ErrRagged
	}
	if len(keys) == 0 {
		return errors.New("no sort keys")
	}
	parsed := make([]sortKey, 0, len(keys))
	for _, k := range keys {
		sk := sortKey{col: strings.TrimPrefix(k, "-"), desc: strings.HasPrefix(k, "-")}
		switch sk.col {
		case "name", "age", "score":
		default:
			return fmt.Errorf("unknown sort key %q", k)
		}
		parsed = append(parsed, sk)
	}
	sort.Stable(byKeys{t, parsed})
	return nil
}
```

**Swap — единственное место, где живёт «строка».** Сортировка вызывает
`Swap(i, j)` и ничего не знает про колонки. Забыть одну колонку в `Swap` —
значит, перемешать данные: баллы уедут к чужим людям, и тест на первой же
сортировке это покажет.

**Убывание — через инверсию сравнения, не результата.** Частая ошибка —
`return !less(i, j)`. Для равных элементов это даёт `true` в обе стороны:
`Less(i, j)` и `Less(j, i)` одновременно истинны, и стабильная сортировка
переставит равные строки. Правильно — поменять знак у трёхзначного
сравнения (или сравнивать `j` с `i`), тогда равные остаются равными.

**NaN ломает строгий порядок.** `NaN < x` и `x < NaN` — оба `false`, то есть
NaN «равен» всем, но числа не равны между собой — транзитивность нарушена, и
результат сортировки непредсказуем. Поэтому NaN обрабатывается явно, до
учёта направления — иначе при убывании NaN уехали бы в начало.

**Сначала проверить все ключи, потом сортировать.** Если разбирать ключи
внутри `Less`, на неизвестном ключе придётся паниковать посреди сортировки, и
таблица останется наполовину переставленной.

Что спросят дальше: почему `byKeys` можно передавать по значению (внутри
указатель и слайсы — `Swap` меняет общие массивы) и как сделать то же через
перестановку индексов: отсортировать `[]int` с `slices.SortStableFunc`, а
потом переложить колонки — это быстрее, когда колонок много.
